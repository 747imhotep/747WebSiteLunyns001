export default {
  async fetch(request: Request): Promise<Response> {
    if (request.method === "OPTIONS") {
      // Handle CORS preflight requests
      return new Response(null, {
        status: 204,
        headers: {
          "Access-Control-Allow-Origin": "*",
          "Access-Control-Allow-Methods": "POST, OPTIONS",
          "Access-Control-Allow-Headers": "Content-Type",
        },
      });
    }

    if (request.method !== "POST") {
      return new Response("Method Not Allowed", {
        status: 405,
        headers: {
          "Access-Control-Allow-Origin": "*",
          "Access-Control-Allow-Methods": "POST, OPTIONS",
          "Access-Control-Allow-Headers": "Content-Type",
        },
      });
    }

    try {
      const body = await request.json();

      // Basic validation
      if (!body.email || !body.name || !body.consent) {
        return new Response("Missing required fields", { status: 400 });
      }

      const senderResponse = await fetch("https://hidden", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${SENDER_API_KEY}`, // Secure environment variable
        },
        body: JSON.stringify({
          email: body.email,
          name: body.name,
          list_ids: ["your-list-id"], // replace with your real list ID
          custom_fields: {
            consent: "true",
          },
        }),
      });

      if (!senderResponse.ok) {
        const error = await senderResponse.text();
        return new Response(`Sender.net API error: ${error}`, { status: 500 });
      }

      return new Response("Subscriber added successfully", {
        status: 200,
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*",
        },
      });
    } catch (err) {
      return new Response(`Server error: ${(err as Error).message}`, {
        status: 500,
        headers: {
          "Access-Control-Allow-Origin": "*",
        },
      });
    }
  },
};
